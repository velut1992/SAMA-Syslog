## Supra Log Collector Configuration (Fluentd)
##
## Collects logs from: IEDs, Switches, Routers, Firewalls,
##                     Windows Workstations/Servers, Linux Servers
##
## Ports:
##    514/UDP+TCP  - IED syslog (timestamps in UTC)
##   1514/UDP      - Windows NXLog agents (JSON format, IST)
##   2514/UDP+TCP  - Network devices: switches, routers, firewalls (timestamps in IST)
##  24224/TCP      - Fluentd forward input (from Fluentd agents)
##
## Port Assignment Guide:
##   IEDs (ABB REC670, REL670, etc.)     → configure syslog to port 514
##   Siemens Ruggedcom switches           → configure syslog to port 2514
##   Routers / Firewalls                  → configure syslog to port 2514
##   Windows servers (NXLog)              → configure NXLog to port 1514
##   Linux servers (rsyslog)              → configure rsyslog to port 514

## ============================================================
## IED PROCESSING (Hitachi Energy REC670 / REL670)
## ============================================================
##
## REC670 syslog message format (pipe-delimited):
##   01:Log-out (user logged out)||1210|01137|AA1D1Q19KF1|REC670|Administrator
##   01:Configuration transferred to the device successfully||13200|01136|AA1D1Q19KF1|REC670|Administrator
##   01:Log-in successful|Accesspoint:1|1110|01135|AA1D1Q19KF1|REC670|Administrator
##
## Fields: priority:description|detail|event_code|sequence|device_id|ied_model|user

# Parse REC670 pipe-delimited message and classify events
# Message format: 01:Description|Detail|Code|Sequence|DeviceID|Model|User
# Example: 01:Log-in successful|Accesspoint:1|1110|01135|AA1D1Q19KF1|REC670|Administrator
<filter ied.**>
  @type record_transformer
  enable_ruby true
  <record>
    event_description ${parts = record["message"].to_s.gsub(/\x00/, "").strip.split("|"); desc = parts[0].to_s; desc.include?(":") ? desc.split(":", 2)[1].to_s.strip : desc}
    event_detail ${parts = record["message"].to_s.gsub(/\x00/, "").strip.split("|"); parts.length >= 2 ? parts[1].to_s.strip : ""}
    event_code ${parts = record["message"].to_s.gsub(/\x00/, "").strip.split("|"); parts.length >= 3 ? parts[2].to_s.strip : ""}
    sequence ${parts = record["message"].to_s.gsub(/\x00/, "").strip.split("|"); parts.length >= 4 ? parts[3].to_s.strip : ""}
    device_id ${parts = record["message"].to_s.gsub(/\x00/, "").strip.split("|"); parts.length >= 5 ? parts[4].to_s.strip : record["host"].to_s}
    ied_model ${parts = record["message"].to_s.gsub(/\x00/, "").strip.split("|"); parts.length >= 6 ? parts[5].to_s.strip : "unknown"}
    user ${parts = record["message"].to_s.gsub(/\x00/, "").strip.split("|"); parts.length >= 7 ? parts[6].to_s.strip : "unknown"}
    event_type ${msg = record["message"].to_s.downcase; msg.include?("trip") ? "trip" : msg.include?("fault") ? "fault" : msg.include?("alarm") ? "alarm" : msg.include?("log-in") ? "login" : msg.include?("log-out") ? "logout" : msg.include?("configuration") ? "config_change" : "info"}
    severity ${msg = record["message"].to_s.downcase; (msg.include?("trip") || msg.include?("fault")) ? "critical" : msg.include?("alarm") ? "warning" : "info"}
    log_collector "supra"
    source_type "ied"
  </record>
</filter>

## ============================================================
## NETWORK (SWITCH) PROCESSING
## ============================================================

<filter network>
  @type record_transformer
  enable_ruby true

  <record>

    switch_name ${
      m = record["message"].to_s;
      m[/\bKC\d+\b/] || "unknown"
    }

    event_type ${
      msg = record["message"].to_s.downcase;

      msg.include?("link down") || msg.include?("interface down") ? "link_down" :
      msg.include?("link up") || msg.include?("interface up") ? "link_up" :

      msg.include?("authentication failed") ? "auth_failed" :
      msg.include?("logged in") ? "login" :
      msg.include?("logged out") ? "logout" :

      msg.include?("configuration") ? "config_change" :

      msg.include?("snmp") ? "snmp_event" :
      msg.include?("ntp") ? "ntp_event" :
      msg.include?("syslog") ? "syslog_event" :

      msg.include?("reboot") || msg.include?("restart") ? "device_restart" :

      "info"
    }

    severity ${
      msg = record["message"].to_s.downcase;

      msg.include?("critical") ||
      msg.include?("failed") ||
      msg.include?("down") ? "critical" :

      msg.include?("warning") ? "warning" :

      "info"
    }

    raw_message ${record["message"]}

    source_type network

    log_collector supra

  </record>
</filter>
