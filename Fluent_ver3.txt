## Supra Log Collector Configuration (Fluentd)
##
## Collects logs from: IEDs, Switches, Routers, Firewalls,
##                     Windows Workstations/Servers, Linux Servers
##
## Ports:
##    514/UDP+TCP  - IED syslog (timestamps in UTC)
##   1514/UDP      - Windows NXLog agents (JSON format, IST)
##   2514/UDP+TCP  - Network devices: switches, routers, firewalls (timestamps in IST)
##  24224/TCP      - Fluentd forward input (from Fluentd agents)
##
## Port Assignment Guide:
##   IEDs (ABB REC670, REL670, etc.)     → configure syslog to port 514
##   Siemens Ruggedcom switches           → configure syslog to port 2514
##   Routers / Firewalls                  → configure syslog to port 2514
##   Windows servers (NXLog)              → configure NXLog to port 1514
##   Linux servers (rsyslog)              → configure rsyslog to port 514

# Windows (Port 1514 - JSON - IST)
<source>
  @type udp
  port 1514
  bind 0.0.0.0
  tag windows
  <parse>
    @type json
    time_key EventTime
    time_format %Y-%m-%d %H:%M:%S
    timezone +05:30
    keep_time_key true
  </parse>
</source>

## ============================================================
## IED PROCESSING (Hitachi Energy REC670 / REL670)
## ============================================================
##
## REC670 syslog message format (pipe-delimited):
##   01:Log-out (user logged out)||1210|01137|AA1D1Q19KF1|REC670|Administrator
##   01:Configuration transferred to the device successfully||13200|01136|AA1D1Q19KF1|REC670|Administrator
##   01:Log-in successful|Accesspoint:1|1110|01135|AA1D1Q19KF1|REC670|Administrator
##
## Fields: priority:description|detail|event_code|sequence|device_id|ied_model|user

# Parse REC670 pipe-delimited message format
<filter ied.**>
  @type parser
  key_name message
  reserve_data true
  emit_invalid_record_to_error false
  <parse>
    @type regexp
    expression /^(?<ied_priority>\d+):(?<event_description>[^|]*)\|(?<event_detail>[^|]*)\|(?<event_code>\d+)\|(?<sequence>\d+)\|(?<device_id>[^|]*)\|(?<ied_model>[^|]*)\|(?<user>[^\x00]*)$/
  </parse>
</filter>

# Event Type and Severity Classification
<filter ied.**>
  @type record_transformer
  enable_ruby true
  <record>
    event_type ${msg = (record["event_description"] || record["message"]).to_s.downcase; msg.include?("trip") ? "trip" : msg.include?("fault") ? "fault" : msg.include?("alarm") ? "alarm" : msg.include?("log-in") ? "login" : msg.include?("log-out") ? "logout" : msg.include?("configuration") ? "config_change" : "info"}
    severity ${msg = (record["event_description"] || record["message"]).to_s.downcase; (msg.include?("trip") || msg.include?("fault")) ? "critical" : msg.include?("alarm") ? "warning" : "info"}
    ied_model ${record["ied_model"] || "unknown"}
    device_id ${record["device_id"] || record["host"] || "unknown"}
    user ${record["user"] || "unknown"}
    log_collector "supra"
    source_type "ied"
  </record>
</filter>

## ============================================================
## NETWORK (SWITCH) PROCESSING
## ============================================================

<filter network.**>
  @type record_transformer
  enable_ruby true
  <record>
    event_type ${msg = record["message"].to_s.downcase; msg.include?("link down") ? "link_down" : msg.include?("link up") ? "link_up" : "info"}
    severity ${record["message"].to_s.downcase.include?("down") ? "critical" : "info"}
    source_type "network"
    log_collector "supra"
  </record>
</filter>

## ============================================================
## WINDOWS PROCESSING
## ============================================================

<filter windows>
  @type record_transformer
  <record>
    source_type "windows"
    log_collector "supra"
  </record>
</filter>
